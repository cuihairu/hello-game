### 1.4 数据持久化挑战

**数据一致性问题**：
```
问题场景：玩家同时在两个设备登录

时间线：
T1: 玩家A在手机登录，获得金币+100
T2: 玩家A在PC登录，获得金币+50
T3: 两个设备同时提交数据到服务器

可能的结果：
  方案1: 最后写入覆盖（丢失数据）
  方案2: 乐观锁（冲突检测）
  方案3: 队列化处理（串行化）

推荐方案：乐观锁 + 版本号
```

**数据持久化策略**：
```python
# 数据持久化管理
class DataPersistenceManager:
    def __init__(self):
        self.write_buffer = {}  # 写缓冲
        self.flush_interval = 5  # 刷新间隔（秒）
    
    def save_player_data(self, player_id, data):
        """保存玩家数据"""
        # 1. 检查版本号
        current_version = self.get_version(player_id)
        if data['version'] != current_version:
            raise VersionConflictError(
                f"数据版本冲突: 期望{current_version}, 实际{data['version']}"
            )
        
        # 2. 写入缓冲
        self.write_buffer[player_id] = {
            'data': data,
            'timestamp': time.time(),
        }
        
        # 3. 如果缓冲满了，立即刷新
        if len(self.write_buffer) >= 1000:
            self.flush_buffer()
    
    def flush_buffer(self):
        """刷新缓冲到数据库"""
        batch = []
        for player_id, item in self.write_buffer.items():
            batch.append({
                'player_id': player_id,
                'data': item['data'],
                'timestamp': item['timestamp'],
            })
        
        # 批量写入数据库
        self.batch_insert(batch)
        
        # 清空缓冲
        self.write_buffer.clear()
    
    def periodic_flush(self):
        """定期刷新"""
        while True:
            time.sleep(self.flush_interval)
            if self.write_buffer:
                self.flush_buffer()
```

class ShareManager:
    def __init__(self):
        self.share_channels = ['world', 'guild', 'team', 'friend']
    
    def share_result(self, player_id, result_type, data, scope):
        """分享游戏结果"""
        share_data = {
            'player_id': player_id,
            'type': result_type,
            'data': data,
            'scope': scope,
            'timestamp': time.time(),
        }
        
        if scope == 'world':
            # 世界频道广播
            self.broadcast_to_world(share_data)
        elif scope == 'guild':
            # 帮会频道广播
            self.broadcast_to_guild(player_id, share_data)
        elif scope == 'team':
            # 队伍频道广播
            self.broadcast_to_team(player_id, share_data)
        elif scope == 'friend':
            # 好友私聊
            self.send_to_friends(player_id, share_data)
    
    def broadcast_to_world(self, data):
        """世界频道广播"""
        # 限制广播频率
        if self.is_rate_limited(data['player_id'], 'world'):
            return
        
        # 构造广播消息
        message = self.format_share_message(data)
        
        # 发送到所有在线玩家
        self.send_to_all_players(message)
    
    def is_rate_limited(self, player_id, channel):
        """检查是否被限流"""
        key = f"share_limit:{player_id}:{channel}"
        count = self.redis.get(key) or 0
        
        if count >= self.get_rate_limit(channel):
            return True
        
        self.redis.incr(key)
        self.redis.expire(key, 60)  # 1分钟窗口
        return False
```

# 聊天消息处理器
class ChatMessageHandler:
    def __init__(self):
        self.message_queue = MessageQueue()
        self.redis = RedisClient()
        self.db = DatabaseClient()
    
    def handle_message(self, player_id, channel, content, target_id=None):
        """处理聊天消息"""
        # 1. 消息过滤
        if not self.message_filter(content):
            return {'error': '消息包含敏感词'}
        
        # 2. 频率限制
        if self.is_rate_limited(player_id, channel):
            return {'error': '发送太频繁，请稍后再试'}
        
        # 3. 构造消息
        message = {
            'id': self.generate_message_id(),
            'player_id': player_id,
            'channel': channel,
            'content': content,
            'timestamp': time.time(),
        }
        
        # 4. 发送消息
        if channel == 'world':
            self.broadcast_world(message)
        elif channel == 'guild':
            self.broadcast_guild(player_id, message)
        elif channel == 'team':
            self.broadcast_team(player_id, message)
        elif channel == 'private':
            self.send_private(player_id, target_id, message)
        
        # 5. 存储历史记录
        if self.should_persist(channel):
            self.save_message(message)
        
        return {'success': True, 'message_id': message['id']}
    
    def message_filter(self, content):
        """消息过滤"""
        # 敏感词过滤
        if self.contains_sensitive_words(content):
            return False
        
        # 广告检测
        if self.detect_advertisement(content):
            return False
        
        # 长度限制
        if len(content) > 500:
            return False
        
        return True
    
    def broadcast_world(self, message):
        """世界频道广播"""
        # 获取所有在线玩家
        online_players = self.redis.get_all_online_players()
        
        # 并行发送消息
        with ThreadPoolExecutor() as executor:
            futures = []
            for player_id in online_players:
                future = executor.submit(
                    self.send_to_player, player_id, message
                )
                futures.append(future)
            
            # 等待所有发送完成
            for future in futures:
                future.result()

### 2.4 敏感词过滤

```python
# 敏感词过滤器
class SensitiveWordFilter:
    def __init__(self):
        self.trie = Trie()
        self.load_sensitive_words()
    
    def load_sensitive_words(self):
        """加载敏感词库"""
        words = self.db.get_sensitive_words()
        for word in words:
            self.trie.insert(word)
    
    def filter(self, text):
        """过滤敏感词"""
        result = list(text)
        i = 0
        
        while i < len(text):
            # 从当前位置开始匹配
            match_length = self.trie.search(text[i:])
            
            if match_length > 0:
                # 替换敏感词
                for j in range(i, i + match_length):
                    result[j] = '*'
                i += match_length
            else:
                i += 1
        
        return ''.join(result)
    
    def contains_sensitive_words(self, text):
        """检查是否包含敏感词"""
        return self.trie.search(text) > 0
```

├── 识别效能瓶颈
├── 分析根本原因
├── 制定改进措施
└── 设定改进目标

Do（执行）
├── 实施改进措施
├── 记录执行过程
├── 收集数据
└── 及时调整

Check（检查）
├── 对比改进前后
├── 分析改进效果
├── 总结经验教训
└── 识别新问题

Act（处理）
├── 标准化有效措施
├── 推广最佳实践
├── 制定下轮计划
└── 持续改进
```

## 总结

研发组织与项目管理是游戏成功的关键因素。核心要点：

1. **策划要专业**：建立规范的策划流程和文档标准
2. **团队要协作**：建立高效的跨团队协作机制
3. **流程要敏捷**：采用适合游戏开发的Scrum实践
4. **分配要合理**：根据能力和任务特点分配工作
5. **技能要提升**：建立持续学习和分享的机制
6. **交接要规范**：建立完整的项目交接流程
7. **度量要科学**：建立研发效能度量体系
8. **改进要持续**：建立PDCA持续改进循环

游戏开发是创意与技术的结合，需要团队的高效协作和持续改进。只有建立完善的研发组织体系，才能保证游戏的品质和交付效率。
